// Paste these rules in:
// Firebase Console → Storage → Rules → Edit

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    // Product images — allow any authenticated user (incl. anonymous) to upload
    match /products/{file} {
      allow read:  if true;
      allow write: if request.auth != null
                   && request.resource.size < 5 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
    }

    // Banner images
    match /banners/{file} {
      allow read:  if true;
      allow write: if request.auth != null
                   && request.resource.size < 5 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
    }

    // Request images (from customers)
    match /requests/{file} {
      allow read:  if request.auth != null;
      allow write: if request.auth != null
                   && request.resource.size < 10 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
    }
  }
}
